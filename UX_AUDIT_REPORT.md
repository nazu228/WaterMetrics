### 🔴 Ключевые болевые точки (Top Friction Points):
1. **Деструктивный сброс данных без подтверждения (Data Loss Risk / Heuristic #5):**
   При смене/перевыборе файла шаблона на Дашборде (`_on_template_selected`) ранее введенные пользователем замены счетчиков в «Мастере замен» безвозвратно затираются (`parent_win.closed_meters = []`), сопровождаясь лишь мимолетным Toast-уведомлением. Если бухгалтер 15 минут вносил 10 замен ИПУ и случайно перетащил файл, его работа теряется без возможности отмены (**Undo / Safe State**).

2. **Отсутствие предиктивной валидации и состояния готовности CTA-кнопки (Heuristic #5 & Fitts's Law):**
   Главная кнопка **«Запустить расчет»** остается активной даже при пустых путях к файлам или некорректных цифрах. Нажатие вызывает раздражающие модальные всплывашки об ошибках (*«Укажите все пути к Excel файлам!»*), перекладывая проверку на пользователя постфактум вместо подсказки *«Что именно не заполнено»* прямо в фокусе внимания.

3. **Когнитивный диссонанс в «Мастере замен счетчиков» (Cognitive Overload / Hick's Law):**

### 🟢 Быстрые победы (Quick Wins — внедрение за 1–2 часа):
- **Smart CTA Auto-State:** динамическая блокировка/разблокировка кнопки «Запустить расчет» с отображением бейджа готовности (*«Готово к расчету»* vs *«Требуется: Файл Аркус, Расход ХВС»*).
- **Inline Health-Check в истории файлов:** проверка существования файлов на диске прямо в таблице истории с визуальными бейджами (зеленая точка — доступен, серая с предупреждением — перемещен/удален).
- **Быстрый расчет прямо в полях ввода:** поддержка простых математических выражений в числовых полях (например, ввод `1200 + 45.5` с автовычислением по `Enter`).
- **Синхронизация активных нормативов на Дашборде:** вывод мини-чипов текущих нормативов (ХВС: 4.04, ГВС: 2.65) в карточку целей с возможностью быстрого редактирования в 1 клик.

### Рецепт 1: Умный Pre-flight Валидатор и Реактивная Кнопка Запуска (UX-1)
**Суть изменения:** Пользователь видит мгновенную причину, почему расчет недоступен, а кнопка CTA визуально отражает готовность системы к 1 клику.

```python
# ui/components/reactive_cta.py
from PySide6.QtWidgets import QPushButton, QToolTip
from PySide6.QtCore import Qt, QTimer
from PySide6.QtGui import QCursor

class SmartRunButton(QPushButton):
    """Кнопка запуска с реактивной валидацией готовности и контекстным тултипом."""
    
    def __init__(self, text: str = "Запустить расчет", parent=None):
        super().__init__(text, parent)
        self.setObjectName("PrimaryButton")
        self._missing_requirements = []
        self.update_state([])

    def update_state(self, missing_reqs: list[str]):
        """missing_reqs: список текстовых описаний того, чего не хватает для запуска."""
        self._missing_requirements = missing_reqs
        is_ready = len(missing_reqs) == 0

        self.setEnabled(is_ready)
        if is_ready:
            self.setText("🚀 Запустить расчет (F5 / Ctrl+R)")
            self.setToolTip("Все данные заполнены корректно. Нажмите для запуска.")
            self.setProperty("ready", True)
        else:
            first_missing = missing_reqs[0]
            self.setText(f"⏳ Требуется: {first_missing}")
            tooltip_msg = "<b>Для запуска расчета необходимо:</b><br/>• " + "<br/>• ".join(missing_reqs)
            self.setToolTip(tooltip_msg)
            self.setProperty("ready", False)

        self.style().unpolish(self)
        self.style().polish(self)
```

```python
# ui/dashboard_page.py (патч метода _on_template_selected)
from PySide6.QtWidgets import QMessageBox

def _on_template_selected(self, tpl_path: str):
    if not tpl_path:
        return

    parent_win = self.window()
    if parent_win:
        closed_count = len(getattr(parent_win, 'closed_meters', []))
        new_count = len(getattr(parent_win, 'new_meters', []))
        total_replacements = closed_count + new_count

        if total_replacements > 0:
            msg_box = QMessageBox(self)
            msg_box.setWindowTitle("Внимание: сброс настроек замен ИПУ")
            msg_box.setIcon(QMessageBox.Icon.Warning)
            msg_box.setText(
                f"У вас настроено <b>{total_replacements}</b> замен приборов учета.<br/>"
                "При смене файла шаблона ведомости эти привязки станут недействительными."
            )
            msg_box.setInformativeText("Вы действительно хотите загрузить новый шаблон и очистить замены?")
            btn_proceed = msg_box.addButton("Загрузить и сбросить", QMessageBox.ButtonRole.AcceptRole)
            btn_cancel = msg_box.addButton("Отмена", QMessageBox.ButtonRole.RejectRole)
            msg_box.setDefaultButton(btn_cancel)

            msg_box.exec()
            if msg_box.clickedButton() != btn_proceed:
                # Откатываем выбор файла
                return

            parent_win.closed_meters = []
            parent_win.new_meters = []
            ToastNotification.show_toast(self, "Предыдущие замены сброшены", "INFO")
```

### Рецепт 3: Продвинутое числовое поле с «живым» вычислением выражений и маской (UX-6)
**Суть изменения:** Позволяет бухгалтеру вводить `1250.5 + 40` прямо в поле ХВС или ГВС, автоматически форматируя результат и исключая необходимость открывать калькулятор Windows.

```python
# ui/components/smart_eval_input.py
import re
from PySide6.QtWidgets import QLineEdit, QSizePolicy
from PySide6.QtCore import Qt, QTimer
from PySide6.QtGui import QKeyEvent, QKeySequence

class SmartMathLineEdit(QLineEdit):
    """Интеллектуальное поле с поддержкой выражений (100+50.5), моноширинным шрифтом и форматированием."""

    def __init__(self, default_val: str = "0.0", parent=None):
        super().__init__(default_val, parent)
        self.setMinimumHeight(34)
        self.setAlignment(Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter)
        self.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Fixed)
        self.returnPressed.connect(self._evaluate_math)
        self.editingFinished.connect(self._evaluate_math)

    def _evaluate_math(self):
        text = self.text().strip().replace(',', '.')
        if not text:
            self.setText("0.0")
            return

        # Безопасная валидация: только цифры, точки и базовые мат. операторы
        if re.match(r'^[0-9\.\+\-\*\/\s\(\)]+$', text):
            try:
                # Вычисление простого выражения
                result = float(eval(text, {"__builtins__": None}, {}))
                if result < 0:
                    result = 0.0
                self.setText(f"{result:.2f}")
                self.setProperty("invalid", False)
            except Exception:
                self.setProperty("invalid", True)
        else:
            self.setProperty("invalid", True)

        self.style().unpolish(self)
        self.style().polish(self)
```

### Рецепт 4: Пошаговый информативный статус расчета в фоне (UX-5)
**Суть изменения:** Превращает «черный ящик» расчета в прозрачный, понятный процесс, снижая тревожность пользователя при обработке крупных таблиц.

```python
# core/calculator_bridge.py
from PySide6.QtCore import QThread, Signal

class CalculationStageWorker(QThread):
    """Фоновый расчет с точной передачей этапов и процента выполнения."""
    
    stage_changed = Signal(int, str, int)  # (current_step, step_name, percent)
    log_signal = Signal(str, str)
    finished_signal = Signal(bool, str)

    def run(self):
        try:
            self.stage_changed.emit(1, "Чтение шаблона и парсинг лицевых счетов...", 15)
            # ... чтение шаблона ...

            self.stage_changed.emit(2, "Сопоставление счетчиков с базой Аркус...", 45)
            # ... парсинг Аркус ...

            self.stage_changed.emit(3, "Расчет баланса, нормативов и ОДН...", 75)
            # ... расчет ...

            self.stage_changed.emit(4, "Форматирование стилей и сохранение книги Excel...", 95)
            # ... сохранение ...

            self.stage_changed.emit(5, "Готово!", 100)
            self.finished_signal.emit(True, "Файл успешно сохранен")
        except Exception as e:
            self.finished_signal.emit(False, str(e))
```
